DATA SEGMENT
MSG1 DB 10,13,"ENTER THE STRING$"
MSG2 DB 10,13,"PALINDROME$"
MSG3 DB 10,13,"NOT PALINDROME$"
STR DB 50 DUP(0)
DATA ENDS

CODE SEGMENT
ASSUME CS:CODE,DS:DATA

START:
MOV AX,DATA
MOV DS,AX

LEA DX,MSG1
MOV AH,09H
INT 21H

LEA SI,STR
MOV AH,01H

READSTRING:
INT 21H
CMP AL,0DH
JE ENDINPUT
MOV [SI],AL
INC SI
JMP READSTRING

ENDINPUT:
MOV AL,'$'
MOV [SI],AL
DEC SI

LEA DI,STR

CHECKPALINDROME:
MOV AL,[DI]
CMP [SI],AL
JNE NOTPALINDROME
INC DI
DEC SI
CMP DI,SI
JL CHECKPALINDROME

PALINDOME:
LEA DX,MSG2
MOV AH,09H
INT 21H
JMP XX

NOTPALINDROME:
LEA DX,MSG3
MOV AH,09H
INT 21H
JMP XX

XX:
MOV AH,4CH
INT 21H

CODE ENDS
END START
